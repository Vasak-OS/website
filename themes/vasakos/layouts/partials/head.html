{{/* Every page used to ship the same and the same description, which is the single worst thing a static site can do to itself in search: forty pages competing for one snippet. Everything below is derived per page, with the site-level values only as a fallback. */}} {{- $isHome := .IsHome -}} {{- $title := cond $isHome (printf "%s — Un sistema operativo libre basado en Arch Linux" site.Title) (printf "%s | %s" .Title site.Title) -}} {{- with .Params.seotitle }}{{ $title = . }}{{ end -}} {{- $description := "" -}} {{- with .Description }}{{ $description = . }} {{- else }}{{ with .Summary }}{{ $description = . }} {{- else }}{{ $description = site.Params.description }}{{ end }}{{ end -}} {{/* .Summary keeps the Markdown line breaks, which end up as literal newlines inside the meta tag and as ragged snippets in the results. */}} {{- $description = $description | plainify | htmlUnescape | replaceRE `\s+` " " | chomp | truncate 158 -}} {{/* A taxonomy term page describes itself; page 2 of anything is not the same page as page 1 and must not claim the same title. */}} {{- if eq .Kind "term" -}} {{- $title = printf "Etiqueta: %s | %s" .Title site.Title -}} {{- $description = printf "Artículos y documentación de VasakOS sobre %s." .Title -}} {{- end -}} {{/* .Paginator errors outright on a page kind that cannot paginate — and *calling* it is what creates the pager, so the home page is left out: its template does not paginate and asking here would generate /page/2..N/ full of duplicate content. */}} {{- if or (eq .Kind "section") (eq .Kind "taxonomy") (eq .Kind "term") -}} {{- with .Paginator -}} {{- if gt .PageNumber 1 -}} {{- $title = printf "%s — Página %d" $title .PageNumber -}} {{- $description = printf "%s (página %d)" $description .PageNumber -}} {{- end -}} {{- end -}} {{- end -}} {{- $image := "/img/og-default.jpg" -}} {{- with .Params.img }}{{ $image = . }}{{ end -}} {{- with .Params.image }}{{ $image = . }}{{ end -}} {{- $image = $image | absURL -}} {{- $type := cond (or .IsHome .IsSection) "website" "article" -}} <head> <meta charset="utf-8" /> <meta name="viewport" content="width=device-width, initial-scale=1" /> <title>{{ $title }} {{/* Taxonomy listings are thin, auto-generated pages: indexing them dilutes the pages that actually answer a query. */}} {{- if or (eq .Kind "taxonomy") (eq .Kind "term") }} {{- end }} {{/* Inline and blocking on purpose: the theme class has to be on before the first paint, or every visitor in dark mode gets a white flash. It is the one script worth blocking on. */}} {{- partial "head/style.html" . -}} {{- if eq $type "article" }} {{- with .Date }}{{ end }} {{- with .Lastmod }}{{ end }} {{- range .Params.tags }}{{ end }} {{- end }} {{- with .OutputFormats.Get "rss" }} {{- end }} {{- partial "head/schema.html" . -}}

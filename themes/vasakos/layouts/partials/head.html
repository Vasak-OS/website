{{/* Everything in the document head, per page and per language. This used to ship the same and the same description on all forty-odd pages, which is the single worst thing a static site can do to itself in search: forty pages competing for one snippet. Everything below is derived per page, with the site-level values only as a fallback. */}} {{- $isHome := .IsHome -}} {{- $lang := .Site.Language -}} {{- $title := cond $isHome (printf "%s — %s" site.Title (T "site.homeTitle")) (printf "%s | %s" .Title site.Title) -}} {{- with .Params.seotitle }}{{ $title = . }}{{ end -}} {{/* El canonical no es siempre `.Permalink`. En una página paginada, Hugo vuelve a renderizar la plantilla de la *sección* para cada tramo, así que `.Permalink` sigue siendo la raíz de la sección: la página 2 se anunciaba como duplicado de la página 1 y Google, que hace caso al canonical, la sacaba del índice. `.Paginator.URL` sí es la URL del tramo. */}} {{- $canonical := .Permalink -}} {{- $paginado := false -}} {{- $description := "" -}} {{- with .Description }}{{ $description = . }} {{- else }}{{ with .Summary }}{{ $description = . }} {{- else }}{{ $description = T "site.summary" }}{{ end }}{{ end -}} {{/* .Summary keeps the Markdown line breaks, which end up as literal newlines inside the meta tag and as ragged snippets in the results. */}} {{- $description = $description | plainify | htmlUnescape | replaceRE `\s+` " " | chomp | truncate 158 -}} {{/* A taxonomy term page describes itself; page 2 of anything is not the same page as page 1 and must not claim the same title. */}} {{- if eq .Kind "term" -}} {{- $title = printf "%s: %s | %s" (T "nav_docs.breadcrumb") .Title site.Title -}} {{- $description = T "head.termDescription" (dict "Term" .Title) -}} {{- end -}} {{/* .Paginator errors outright on a page kind that cannot paginate — and *calling* it is what creates the pager, so the home page is left out: its template does not paginate and asking here would generate /page/2..N/ full of duplicate content. */}} {{- if or (eq .Kind "section") (eq .Kind "taxonomy") (eq .Kind "term") -}} {{- with .Paginator -}} {{- if gt .PageNumber 1 -}} {{- $title = printf "%s — %s %d" $title (T "pagination.page" | default "Página") .PageNumber -}} {{- $description = printf "%s (%s %d)" $description (T "pagination.page" | default "Página") .PageNumber -}} {{- $canonical = .URL | absURL -}} {{- $paginado = true -}} {{- end -}} {{- end -}} {{- end -}} {{- $image := "/img/og-default.jpg" -}} {{- with .Params.img }}{{ $image = . }}{{ end -}} {{- with .Params.image }}{{ $image = . }}{{ end -}} {{- $image = $image | absURL -}} {{- $type := cond (or .IsHome .IsSection) "website" "article" -}} <head> <meta charset="utf-8" /> <meta name="viewport" content="width=device-width, initial-scale=1" /> <title>{{ $title }} {{/* Canonical is per language, so the Spanish site does not tell Google that its English twin is a duplicate of it. With two languages, the canonical of a page in `/en/` points inside `/en/`. On a paginated page it is the URL of that page of the list, not the root of the section. */}} {{/* hreflang. `.AllTranslations` only lists the pages that exist in more than one language, so a page with no English counterpart is not advertised as having one. `x-default` points at the Spanish version, because that is the one whose URL has been published the longest and the one most links aim at. A paginated page is left out on purpose. It is a continuation of a list, not a document with versions in two languages, and `.AllTranslations` hands back the *section* of each language -- so the page 2 of the blog was announcing itself as the Spanish translation of `/en/blog/`, while its own canonical said it was `/blog/page/2/`. Two contradictory declarations on the same page, and both of them wrong. */}} {{- if not $paginado }} {{- range .AllTranslations }} {{- end }} {{- with .AllTranslations }} {{- range . }} {{- if eq .Site.Language.Lang site.Home.Lang }} {{- end }} {{- end }} {{- end }} {{- end }} {{/* Taxonomy listings are thin, auto-generated pages: indexing them dilutes the pages that actually answer a query. */}} {{- if or (eq .Kind "taxonomy") (eq .Kind "term") }} {{- end }} {{/* Inline and blocking on purpose: the theme class has to be on before the first paint, or every visitor in dark mode gets a white flash. It is the one script worth blocking on. */}} {{/* The hero picture is the home page's LCP element. Telling the browser to start fetching it in the buys back the time that used to be lost discovering the in the body. Same resource pipeline hero.html uses, so the bytes fetched are the bytes the browser ends up picking. */}} {{- if .IsHome }} {{- with resources.Get "img/hero.jpg" }} {{- $srcset := (slice (.Resize "1024x webp q72") (.Resize "1920x webp q72")) }} {{- with (.Resize "1600x jpg q78") }} {{- end }} {{- end }} {{- end }} {{- partial "head/style.html" . -}} {{/* og:locale wants the underscore form, and it changes with the language: telling a Spanish page's crawler that it is en_US makes the wrong translation the one that gets shared. */}} {{/* `og:locale:alternate` son las *otras* versiones de la página. Con `.AllTranslations` cada página se anunciaba a sí misma como su propia alternativa, que es ruido: el que comparte en Facebook se anuncia como disponible en un idioma que ya es el suyo. `.Translations` excluye la página actual. */}} {{- range .Translations }} {{- end }} {{- if eq $type "article" }} {{- with .Date }}{{ end }} {{- with .Lastmod }}{{ end }} {{- range .Params.tags }}{{ end }} {{- end }} {{- with .OutputFormats.Get "rss" }} {{- end }} {{- partial "head/schema.html" . -}}

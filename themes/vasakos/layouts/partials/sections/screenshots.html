{{/* Screenshot gallery. The images live in `static/`, so Hugo cannot resize them and cannot know whether a `.webp` sibling exists. Each entry names its `.jpg` and the `` below is built on that one assumption: a `.webp` of the same name sits next to it. A `` pointing at a `.webp` that was never generated is exactly the bug this had once — a browser that supports WebP would prefer it and then show nothing. `scripts/optimize-images.mjs` writes both files together for that reason. The markup is a real ``, not something a script injects: these are the pictures of the product, and a visitor without JavaScript — or a crawler — should still see them. The script below is only the carousel. */}} {{- $cfg := site.Params.images.screenshots | default dict -}} {{- $shots := slice -}} {{- range ($cfg.main | default slice) -}} {{- if .image -}} {{- $shots = $shots | append . -}} {{- end -}} {{- end -}} {{- if and ($cfg.enabled | default false) $shots -}} {{- /* The keys the captions and alt text are looked up by. They come from i18n/.toml, not from params.yaml, so a screenshot has one description in one place instead of two that drift apart. */ -}} {{- $id := . -}}
{{ T "screenshots.eyebrow" }}

{{ T "screenshots.title" }}

{{ T "screenshots.subtitle" }}

{{- range $index, $ss := $shots }} {{- $key := printf "screenshots.%s" $ss.identifier }} {{- $alt := T (printf "%s.alt" $key) | default (T "screenshots.defaultAlt") }} {{- $caption := T (printf "%s.caption" $key) }} {{- /* `image` trae el ancho pegado al final (`shot-01-1536.jpg`), y el `srcset` lo necesita sin él: los archivos se llaman `shot-01-768.webp`, no `shot-01-1536-768.webp`. */ -}} {{- $stem := (strings.TrimSuffix ".jpg" $ss.image) | replaceRE `-\d+$` "" -}} {{- $w := $ss.width | default 1536 }} {{- $h := $ss.height | default 836 }} {{- /* Un `srcset` por formato. La tarjeta no pasa de 768 px, así que sin esto el navegador estaría reduciendo siempre la imagen grande, y el reescalado es justo lo que se ve borroso. */ -}} {{- $srcsetWebp := slice -}} {{- $srcsetJpg := slice -}} {{- range ($ss.srcset | default (slice 768 1536)) -}} {{- $srcsetWebp = $srcsetWebp | append (printf "%s-%d.webp %dw" $stem . .) -}} {{- $srcsetJpg = $srcsetJpg | append (printf "%s-%d.jpg %dw" $stem . .) -}} {{- end -}} {{- end }}
